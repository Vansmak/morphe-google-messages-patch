/**
 * A context for patches containing the current state of resources.
 *
 * Resources are decoded when patches request them, instead of all before patches execute. The manifest
 * is always decoded. A file a patch reads by its path with [get] or [document] is decoded then, and
 * [decode] decodes every file matching a pattern, which a patch that lists a directory must call first,
 * since a directory only holds the files decoded so far.
 * [listDecodableFiles] lists what can be decoded without decoding it.
 */
class ResourcePatchContext internal constructor(
    private val config: PatcherConfig,
) : PatchContext<PatcherResult.PatchedResources>, Closeable {
    private val logger = Logger.getLogger(ResourcePatchContext::class.java.name)

    override val fileWorkspace = config.fileWorkspace

    override val validate = config.validate

    private val resourceCoder: ResourceCoder = ArsclibResourceCoder(config.apkFiles, config.apkFile, config.keepArchitectures)

    /**
     * The resource ids of the APK being patched, read from its resource table on first use.
     * It is also reachable without a context through [app.morphe.patcher.resource.resourceId] and
     * [app.morphe.patcher.resource.hasResourceId], which is what [app.morphe.patcher.resourceLiteral] uses.
     */
    val resourceIds: ResourceIds = ResourceIds { resourceCoder.resourceIds() }.also { ResourceIds.current = it }

    val packageMetadata = resourceCoder.getPackageMetadata()

    /**
     * Read a document from an [InputStream].
     */
    @Suppress("unused")
    fun document(inputStream: InputStream) = Document(inputStream)

    /**
     * Read and write documents in the [PatcherConfig.apkFiles].
     * The file is decoded first if it is not decoded yet.
     */
    @Suppress("unused")
    fun document(path: String) = Document(resourceCoder.getFile(path))
    @Suppress("unused")
    fun document(path: String, packageName: String) = Document(resourceCoder.getFile(path, packageName))

    /**
     * Decode resources of [PatcherConfig.apkFile].
     *
     * @param mode The [ResourceMode] to use.
     */
    internal fun decodeResources(mode: ResourceMode): PackageMetadata {
        config.initializeTemporaryFilesDirectories()
        val onRequest = config.uses(PatcherBehavior.ON_REQUEST_RESOURCE_DECODING)
        if (mode == ResourceMode.FULL) {
            logger.info(if (onRequest) "Decoding the manifest" else "Decoding all resources")
            return resourceCoder.decodeResources(onRequest)
        } else {
            logger.info(if (onRequest) "Extracting raw resources on request" else "Decoding resources in raw mode")
            return resourceCoder.decodeRaw(onRequest)
        }
    }

    /**
     * Called once patches no longer request resources, which releases the resource table on small heaps
     * before the DEX files are written.
     */
    internal fun finishDecoding() = resourceCoder.finishDecoding()

    /**
     * Whether the resource APK [get] writes leaves out an archive entry, as the patched APK replaces or
     * deletes it. Set once the DEX files are written.
     */
    internal var omitEntry: (String) -> Boolean = { false }

    /**
     * Compile resources in [PatcherConfig.apkFiles].
     *
     * @return The [PatcherResult.PatchedResources].
     */
    @InternalApi
    override fun get(): PatcherResult.PatchedResources {
        val resourcesApkFile = if (config.resourceMode == ResourceMode.FULL) {
            logger.info("Compiling modified resources")
            resourceCoder.encodeResources(config.patchedFiles, omitEntry)
        } else {
            null
        }

        // FIXME: All of this stuff is handled by arsclib using metadata files. Clean this up.
        return PatcherResult.PatchedResources(
            resourcesApkFile,
            resourceCoder.getOtherResourceFiles(config.patchedFiles, config.resourceMode),
            resourceCoder.getUncompressedFiles(config.resourceMode),
            resourceCoder.getDeletedFiles(config.resourceMode),
        )
    }

    /**
     * Compile resources in [PatcherConfig.apkFiles] and write the patched APK with [dexFiles], as [get]
     * followed by [app.morphe.patcher.apk.ApkUtils.applyTo] would.
     *
     * @param outputApk The file to write the patched APK to.
     * @param dexFiles The patched DEX files, compressed in the background.
     * @param timings Where the steps of writing the APK are measured.
     */
    internal fun writeApk(outputApk: File, dexFiles: CompressedDexFiles, timings: PhaseTimings) {
        if (config.resourceMode == ResourceMode.FULL) logger.info("Compiling modified resources")
        resourceCoder.writeApk(outputApk, config.patchedFiles, config.resourceMode, omitEntry, dexFiles, timings)
    }

    /**
     * Get a file from [PatcherConfig.apkFiles].
     *
     * A directory only holds the files that were decoded or added so far, so call [decode] before listing it.
     *
     * @param path The path of the file.
     * @param copy Whether to decode the file from [PatcherConfig.apkFile] if it is not decoded yet. Pass false
     * to replace or add a file without decoding what it replaces. A values file written this way, such as
     * `res/values-de/strings.xml`, adds its resources to those of the APK or replaces the ones of the same
     * name, but removes none.
     */
    operator fun get(
        path: String,
        copy: Boolean = true,
    ) = resourceCoder.getFile(path, copy = copy)

    /**
     * List the files of the APK being patched that can be decoded, by the path they are decoded to,
     * without decoding them. These are the manifest, the resource files under `res/` and the values files
     * `resources.arsc` is decoded to, such as `res/values-de/strings.xml` and `res/values/public.xml`, and
     * the other entries of the APK, such as `assets/data.bin`, which are extracted as they are. Native
     * libraries and DEX files are not listed, but [get] extracts a native library all the same.
     *
     * The listing describes the input APK, so it does not reflect files a patch has since added or deleted.
     *
     * @param pattern A glob pattern the paths must match. `*` matches any characters within one segment of
     * a path, `**` matches any characters across segments, and `?` matches one character. A pattern that
     * matches a directory matches everything below it, so `res/xml` lists the files in that directory.
     * Lists every file by default.
     * @param packageName The package whose resources to list. Defaults to the package name of the APK.
     * @return The matching paths, sorted.
     */
    fun listDecodableFiles(pattern: String = "**", packageName: String? = null) =
        resourceCoder.listDecodableFiles(pattern, packageName)

    /**
     * Decode the files of the APK being patched that match [patterns] and are not decoded yet, so that a
     * patch can find them in [PatcherConfig.apkFiles], for example to list a directory.
     *
     * A file that was decoded before, or that a patch changed, added or deleted, is left as it is. A values
     * file a patch wrote before it was decoded is decoded and keeps what the patch wrote, which replaces the
     * resources of the same name. Calling this when every resource was decoded up front does nothing more.
     *
     * ```kt
     * // Every layout of every configuration.
     * decode("res/layout*").forEach { path ->
     *     document(path).use { document -> }
     * }
     * ```
     *
     * @param patterns Glob patterns the paths must match, see [listDecodableFiles]. `**` decodes everything.
     * @param packageName The package whose resources to decode. Defaults to the package name of the APK.
     * @return The paths of the files in [PatcherConfig.apkFiles] matching [patterns], sorted. These include
     * files patches added, but not the ones they deleted.
     */
    fun decode(vararg patterns: String, packageName: String? = null) =
        resourceCoder.decode(patterns.toList(), packageName)

    /**
     * List the entries of the APK being patched, without staging them to the working directory.
     *
     * Use this to discover what an APK contains, rather than walking the working directory: not
     * every entry is staged there, and native libraries in particular are left in the archive.
     * Names are archive names, so they can be passed straight to [get].
     *
     * The listing describes the input APK, so it does not reflect files a patch has since added,
     * changed or deleted.
     *
     * @param prefix Restricts the listing to entries starting with it, e.g. `lib/`.
     */
    fun listApkEntries(prefix: String = "") = resourceCoder.listApkEntries(prefix)

    /**
     * Mark a file for deletion when the APK is rebuilt.
     *
     * The name is either a decoded resource path such as `res/drawable/icon.png`, or any other
     * entry of the APK being patched as returned by [listApkEntries], such as
     * `lib/x86/libfoo.so` or `assets/data.bin`. Archive entries are excluded from the rebuilt APK
     * staged to the working directory or not, so this works for native
     * libraries too. A directory name, such as `lib/x86/`, deletes everything below it. Deleting a
     * name that matches nothing is a no-op. A file deleted and then recreated with [get] keeps the
     * recreated content. The manifest and `resources.arsc` cannot be deleted.
     *
     * @param name The name of the file to delete.
     * @param packageName The package name a decoded resource exists in. Defaults to the package name of the APK.
     */
    @Suppress("unused")
    fun delete(name: String, packageName: String? = null) = resourceCoder.deleteFile(name, packageName)

    override fun close() {
        if (ResourceIds.current === resourceIds) ResourceIds.current = null
        resourceCoder.close()
    }

class Document private constructor(
    document: Document,
    private val file: File?,
) : Document by document, Closeable {
    init {
        normalize()
    }

    internal constructor(inputStream: InputStream) : this(parse(inputStream), null)

    internal constructor(file: File) : this(file.inputStream().buffered().use(::parse), file) {
        readerCount.merge(file, 1, Int::plus)
    }

    override fun close() {
        file?.let {
            if (readerCount[it]!! > 1) {
                throw IllegalStateException(
                    "Two or more instances are currently reading $it." +
                        "To be able to close this instance, no other instances may be reading $it at the same time.",
                )
            } else {
                readerCount.remove(it)
            }

            it.utf8Writer().use { writer -> XmlWriter(writer).writeDocument(this) }
        }
    }

    private companion object {
        private val readerCount = mutableMapOf<File, Int>()

        fun parse(inputStream: InputStream): Document =
            DocumentBuilderFactory.newInstance().newDocumentBuilder().parse(inputStream).apply {
                documentElement?.resolveAttributeNamespaces(emptyMap())
            }

        /**
         * Gives each prefixed attribute the namespace its prefix is declared with, as a namespace aware
         * parser would, so that the namespace methods such as [Element.setAttributeNS] replace a parsed
         * attribute instead of adding a second attribute with the same name.
         *
         * The parser is not namespace aware, because a namespace aware parser fails on a prefix that is
         * not declared, and on Android leaves the `xmlns` attributes out of the document.
         *
         * @param namespaces The namespace URI each prefix is declared with at the parent element.
         */
        private fun Element.resolveAttributeNamespaces(namespaces: Map<String, String>) {
            val attributeList = (0 until attributes.length).map { attributes.item(it) as Attr }

            var elementNamespaces = namespaces
            attributeList.filter { it.name.startsWith("xmlns:") }.takeIf { it.isNotEmpty() }?.let { declarations ->
                elementNamespaces = namespaces + declarations.map { it.name.substring("xmlns:".length) to it.value }
            }

            if (elementNamespaces.isNotEmpty()) {
                val namespaceURIs = attributeList.map { attribute ->
                    val prefix = attribute.name.substringBefore(':', "")
                    if (prefix == "xmlns") null else elementNamespaces[prefix]
                }

                // All attributes are added again in their order, which adding only the replaced ones would change.
                if (namespaceURIs.any { it != null }) {
                    attributeList.forEach(::removeAttributeNode)
                    attributeList.forEachIndexed { index, attribute ->
                        val namespaceURI = namespaceURIs[index]
                        if (namespaceURI == null) {
                            setAttributeNode(attribute)
                        } else {
                            setAttributeNodeNS(
                                ownerDocument.createAttributeNS(namespaceURI, attribute.name).also {
                                    it.value = attribute.value
                                },
                            )
                        }
                    }
                }
            }

            var child = firstChild
            while (child != null) {
                if (child is Element) child.resolveAttributeNamespaces(elementNamespaces)
                child = child.nextSibling
            }
        }
    }
}

/**
 * Writes a DOM document as UTF-8 XML.
 *
 * This replaces the platform transformer, which on Android checks every character against the output
 * encoding and takes a lock for every character it writes, and which escapes surrogate pairs to
 * invalid character references unless it writes UTF-16.
 *
 * Namespace declarations missing from the document are added where they are first used, such as for
 * nodes a patch adds in a namespace the document does not declare.
 */
internal class XmlWriter(private val writer: Writer) {
    /**
     * The namespace URI each prefix is bound to at the element being written.
     * The default namespace has the empty prefix.
     */
    private val boundNamespaces = mutableMapOf<String, String>()

    fun writeDocument(document: Document) {
        writer.write("<?xml version=\"1.0\" encoding=\"UTF-8\"")
        if (document.xmlStandalone) writer.write(" standalone=\"yes\"")
        writer.write("?>")
        var child = document.firstChild
        while (child != null) {
            writeNode(child)
            child = child.nextSibling
        }
    }

    private fun writeNode(node: Node) {
        when (node.nodeType) {
            Node.ELEMENT_NODE -> writeElement(node as Element)
            Node.TEXT_NODE -> writeEscaped(node.nodeValue, inAttribute = false)
            Node.CDATA_SECTION_NODE -> {
                // "]]>" cannot be inside a CDATA section, so it is split into two sections.
                writer.write("<![CDATA[")
                writer.write(node.nodeValue.replace("]]>", "]]]]><![CDATA[>"))
                writer.write("]]>")
            }

            Node.COMMENT_NODE -> {
                writer.write("<!--")
                writer.write(node.nodeValue)
                writer.write("-->")
            }

            Node.PROCESSING_INSTRUCTION_NODE -> {
                val instruction = node as ProcessingInstruction
                writer.write("<?")
                writer.write(instruction.target)
                if (!instruction.data.isNullOrEmpty()) {
                    writer.write(" ")
                    writer.write(instruction.data)
                }
                writer.write("?>")
            }

            Node.DOCUMENT_TYPE_NODE -> writeDocumentType(node as DocumentType)

            Node.ENTITY_REFERENCE_NODE -> {
                writer.write("&")
                writer.write(node.nodeName)
                writer.write(";")
            }

            else -> throw IllegalStateException("Cannot write XML node of type ${node.nodeType}")
        }
    }

    private fun writeElement(element: Element) {
        writer.write("<")
        writer.write(element.tagName)

        // The previous bindings of the prefixes this element binds, restored when the element ends.
        val previousBindings = mutableListOf<Pair<String, String?>>()
        fun bind(prefix: String, namespaceURI: String) {
            previousBindings += prefix to boundNamespaces.put(prefix, namespaceURI)
        }

        val attributes = element.attributes
        for (index in 0 until attributes.length) {
            val attribute = attributes.item(index) as Attr
            if (attribute.name == "xmlns") {
                bind("", attribute.value)
            } else if (attribute.name.startsWith("xmlns:")) {
                bind(attribute.name.substring("xmlns:".length), attribute.value)
            }
        }

        fun declareIfUnbound(prefix: String?, namespaceURI: String?) {
            if (namespaceURI == null || prefix == "xml" || namespaceURI == XMLNS_NAMESPACE_URI) return
            val boundPrefix = prefix.orEmpty()
            if (boundNamespaces[boundPrefix] == namespaceURI) return

            bind(boundPrefix, namespaceURI)
            writer.write(if (boundPrefix.isEmpty()) " xmlns=\"" else " xmlns:$boundPrefix=\"")
            writeEscaped(namespaceURI, inAttribute = true)
            writer.write("\"")
        }

        declareIfUnbound(element.prefix, element.namespaceURI)
        for (index in 0 until attributes.length) {
            val attribute = attributes.item(index) as Attr
            // An attribute without a prefix is in no namespace, whatever the default namespace is.
            if (attribute.prefix != null) declareIfUnbound(attribute.prefix, attribute.namespaceURI)
        }

        for (index in 0 until attributes.length) {
            val attribute = attributes.item(index) as Attr
            writer.write(" ")
            writer.write(attribute.name)
            writer.write("=\"")
            writeEscaped(attribute.value, inAttribute = true)
            writer.write("\"")
        }

        var child = element.firstChild
        if (child == null) {
            writer.write("/>")
        } else {
            writer.write(">")
            while (child != null) {
                writeNode(child)
                child = child.nextSibling
            }
            writer.write("</")
            writer.write(element.tagName)
            writer.write(">")
        }

        for ((prefix, namespaceURI) in previousBindings.asReversed()) {
            if (namespaceURI == null) boundNamespaces.remove(prefix) else boundNamespaces[prefix] = namespaceURI
        }
    }

    private fun writeDocumentType(documentType: DocumentType) {
        writer.write("<!DOCTYPE ")
        writer.write(documentType.name)
        val publicId = documentType.publicId
        val systemId = documentType.systemId
        if (!publicId.isNullOrEmpty()) {
            writer.write(" PUBLIC \"$publicId\" \"${systemId.orEmpty()}\"")
        } else if (!systemId.isNullOrEmpty()) {
            writer.write(" SYSTEM \"$systemId\"")
        }
        if (!documentType.internalSubset.isNullOrEmpty()) {
            writer.write(" [")
            writer.write(documentType.internalSubset)
            writer.write("]")
        }
        writer.write(">")
    }

    /**
     * Writes [text] with the characters escaped that XML requires, writing the runs between them at once.
     * In an attribute, white space other than a space is also escaped, which a parser would otherwise
     * normalize to a space.
     */
    private fun writeEscaped(text: String, inAttribute: Boolean) {
        var runStart = 0
        for (index in text.indices) {
            val replacement = when (text[index]) {
                '&' -> "&amp;"
                '<' -> "&lt;"
                '>' -> "&gt;"
                '"' -> if (inAttribute) "&quot;" else null
                '\n' -> if (inAttribute) "&#10;" else null
                '\t' -> if (inAttribute) "&#9;" else null
                '\r' -> "&#13;"
                else -> null
            } ?: continue

            if (index > runStart) writer.write(text, runStart, index - runStart)
            writer.write(replacement)
            runStart = index + 1
        }
        if (runStart < text.length) writer.write(text, runStart, text.length - runStart)
    }

    private companion object {
        private const val XMLNS_NAMESPACE_URI = "http://www.w3.org/2000/xmlns/"
    }
}
